#!/bin/bash # ============================================================================== # Universal PXE Boot Server Setup Script (Production Grade) # Features: Multi-Distro Selection, State Detection, Append-Mode, # Security (MAC Whitelist + Password), Auto-Recovery # ============================================================================== # Colors GREEN='\033[0;32m' BLUE='\033[0;34m' YELLOW='\033[1;33m' RED='\033[0;31m' CYAN='\033[0;36m' BOLD='\033[1m' NC='\033[0m' # Logging LOG_FILE="/var/log/pxe_setup.log" log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"; } info() { echo -e "${BLUE}[INFO]${NC} $1"; log "INFO: $1"; } success() { echo -e "${GREEN}[ OK ]${NC} $1"; log "OK: $1"; } warn() { echo -e "${YELLOW}[WARN]${NC} $1"; log "WARN: $1"; } fail() { echo -e "${RED}[FAILED]${NC} $1"; log "FAIL: $1"; } die() { fail "$1"; echo -e "${RED}Setup aborted. Check log: $LOG_FILE${NC}"; exit 1; } run() { local description="$1" shift info "$description" log "EXEC: $*" if "$@" >> "$LOG_FILE" 2>&1; then success "$description" return 0 else fail "$description" return 1 fi } verify_file() { if [ -f "$1" ]; then success "Verified: $2 exists"; return 0; else fail "Verification failed: $2 missing"; return 1; fi } verify_dir() { if [ -d "$1" ]; then success "Verified: $2 exists"; return 0; else fail "Verification failed: $2 missing"; return 1; fi } verify_service() { if systemctl is-active --quiet "$1" 2>/dev/null; then success "Service '$1' running"; return 0; else fail "Service '$1' NOT running"; return 1; fi } cleanup() { mountpoint -q /mnt/iso 2>/dev/null && umount /mnt/iso 2>/dev/null; } trap cleanup EXIT echo -e "${CYAN}======================================================${NC}" echo -e "${GREEN} Universal PXE Boot Server Setup (v2.0)${NC}" echo -e "${CYAN}======================================================${NC}" if [[ $EUID -ne 0 ]]; then die "This script must be run as root. Try: sudo $0"; fi PI_IP=$(hostname -I | awk '{print $1}') if [ -z "$PI_IP" ]; then die "Could not detect a valid IP address."; fi SUBNET_PREFIX=$(echo "$PI_IP" | cut -d. -f1,2,3) SUBNET="${SUBNET_PREFIX}.0" HOSTNAME=$(hostname) # ==================== OS CATALOG ==================== OS_NAMES=("Linux Mint 21.3 XFCE" "Linux Mint 21.3 Cinnamon" "Ubuntu 24.04 Desktop" "Ubuntu 22.04 Desktop" "Debian 12 XFCE Live" "Arch Linux (Latest)") OS_URLS=( "https://mirrors.kernel.org/linuxmint/stable/21.3/linuxmint-21.3-xfce-64bit.iso" "https://mirrors.kernel.org/linuxmint/stable/21.3/linuxmint-21.3-cinnamon-64bit.iso" "https://releases.ubuntu.com/24.04.1/ubuntu-24.04.1-desktop-amd64.iso" "https://releases.ubuntu.com/22.04.5/ubuntu-22.04.5-desktop-amd64.iso" "https://cdimage.debian.org/debian-cd/current-live/amd64/iso-hybrid/debian-live-12.7.0-amd64-xfce.iso" "https://geo.mirror.pkgbuild.com/iso/latest/archlinux-x86_64.iso" ) OS_ISOS=("mint213_xfce.iso" "mint213_cin.iso" "ubuntu2404.iso" "ubuntu2204.iso" "debian12.iso" "arch.iso") OS_DIRS=("mint213_xfce" "mint213_cin" "ubuntu2404" "ubuntu2204" "debian12" "arch") OS_TYPES=("casper" "casper" "casper" "casper" "debian-live" "arch") # ==================== STATE DETECTION ==================== SETUP_STATE="NEW" if [ -f /srv/tftp/pxelinux.cfg/default ] && [ -d /var/www/html ]; then SETUP_STATE="EXISTING" echo -e "${GREEN}[*] Existing PXE network boot setup detected on this server!${NC}" echo "" fi # ==================== MAIN MENU ==================== while true; do if [ "$SETUP_STATE" == "EXISTING" ]; then echo -e "${BOLD}${YELLOW}What would you like to do?${NC}" echo " 1) Add more Distros to existing PXE Menu" echo " 2) Wipe & Reinstall PXE Setup completely" echo -e " ${RED}3) Uninstall completely & Clean Up${NC}" echo " 4) Exit" read -p "Enter choice [1-4]: " ACTION_CHOICE /dev/null apt purge -y dnsmasq apache2 pxelinux syslinux-common aria2 &>/dev/null apt autoremove -y &>/dev/null rm -rf /srv/tftp /var/www/html/*.iso sed -i '/loop,ro/d' /etc/fstab 2>/dev/null # Remove loop mounts echo -e "${GREEN}Uninstallation complete!${NC}"; exit 0 else echo "Aborted."; exit 0 fi fi ;; 4) echo "Exiting."; exit 0 ;; *) echo -e "${RED}Invalid choice.${NC}" ;; esac done # ==================== DISTRO SELECTION ==================== echo "" echo -e "${BOLD}${CYAN}Available Operating Systems:${NC}" for i in "${!OS_NAMES[@]}"; do echo " $((i+1))) ${OS_NAMES[$i]}" done echo "" echo "You can select multiple operating systems by separating them with spaces." read -p "Enter the numbers you want to install (e.g., '1 3 5'): " -a SELECTED_DISTROS /dev/null apt install -y dnsmasq apache2 pxelinux syslinux-common aria2 wget &>/dev/null || die "Package install failed" sed -i 's/Listen 80$/Listen 8080/' /etc/apache2/ports.conf mv /var/www/html/index.html /var/www/html/index.html.backup 2>/dev/null || true systemctl restart apache2 mkdir -p /srv/tftp/pxelinux.cfg cd /tmp; rm -f netboot.tar.gz wget -q https://deb.debian.org/debian/dists/bookworm/main/installer-amd64/current/images/netboot/netboot.tar.gz tar -xzf netboot.tar.gz cp /tmp/debian-installer/amd64/pxelinux.0 /srv/tftp/ cp /tmp/debian-installer/amd64/boot-screens/*.c32 /srv/tftp/ rm -rf /tmp/debian-installer /tmp/netboot.tar.gz fi # ==================== DOWNLOAD & EXTRACT LOOP ==================== echo -e "\n${BLUE}--- Downloading and Preparing Operating Systems ---${NC}" cd /var/www/html for idx in "${VALID_SELECTIONS[@]}"; do NAME="${OS_NAMES[$idx]}" URL="${OS_URLS[$idx]}" ISO="${OS_ISOS[$idx]}" DIR="${OS_DIRS[$idx]}" TYPE="${OS_TYPES[$idx]}" echo -e "\n${BOLD}${CYAN}Processing: $NAME${NC}" if [ -f "/var/www/html/$ISO" ]; then success "ISO already exists: $ISO" else info "Downloading $ISO..." aria2c -x 16 -s 16 "$URL" -o "$ISO" || die "Failed to download $NAME" fi chmod 644 "/var/www/html/$ISO" mkdir -p /srv/tftp/$DIR mkdir -p /mnt/iso umount /mnt/iso 2>/dev/null || true mount -o loop /var/www/html/$ISO /mnt/iso || die "Failed to mount $ISO" # Extract kernel and initrd based on type if [ "$TYPE" == "arch" ]; then # Arch needs the whole ISO mounted persistently for HTTP rootfs fetching mkdir -p /var/www/html/$DIR if ! grep -q "/var/www/html/$ISO" /etc/fstab; then echo "/var/www/html/$ISO /var/www/html/$DIR udf,iso9660 loop,ro 0 0" >> /etc/fstab fi mount /var/www/html/$DIR 2>/dev/null || true cp /var/www/html/$DIR/arch/boot/x86_64/vmlinuz-linux /srv/tftp/$DIR/vmlinuz cp /var/www/html/$DIR/arch/boot/x86_64/initramfs-linux.img /srv/tftp/$DIR/initrd elif [ "$TYPE" == "casper" ]; then cp /mnt/iso/casper/vmlinuz /srv/tftp/$DIR/ if [ -f /mnt/iso/casper/initrd.lz ]; then cp /mnt/iso/casper/initrd.lz /srv/tftp/$DIR/initrd; elif [ -f /mnt/iso/casper/initrd ]; then cp /mnt/iso/casper/initrd /srv/tftp/$DIR/initrd; else die "Initrd not found in casper for $NAME"; fi elif [ "$TYPE" == "debian-live" ]; then cp /mnt/iso/live/vmlinuz-* /srv/tftp/$DIR/vmlinuz cp /mnt/iso/live/initrd.img-* /srv/tftp/$DIR/initrd fi umount /mnt/iso 2>/dev/null || true success "Kernel & Initrd extracted to /srv/tftp/$DIR/" done # ==================== BUILD PXE MENU ==================== echo -e "\n${BLUE}--- Rebuilding PXE Menu ---${NC}" if [ "$SETUP_STATE" == "NEW" ]; then cat < /srv/tftp/pxelinux.cfg/default DEFAULT vesamenu.c32 PROMPT 0 TIMEOUT 300 MENU TITLE PXE Network Boot Server ($HOSTNAME) LABEL local MENU LABEL Boot from Local Hard Drive LOCALBOOT 0 EOF fi PASSWORD_LINE="" if [[ "$ENABLE_MENU_PASSWORD" == "y" || "$ENABLE_MENU_PASSWORD" == "Y" ]]; then PASSWORD_LINE=" MENU PASSWD $MENU_PASSWORD" fi for idx in "${VALID_SELECTIONS[@]}"; do NAME="${OS_NAMES[$idx]}" ISO="${OS_ISOS[$idx]}" DIR="${OS_DIRS[$idx]}" TYPE="${OS_TYPES[$idx]}" # Check if entry already exists if grep -q "LABEL $DIR" /srv/tftp/pxelinux.cfg/default; then warn "Menu entry for $NAME already exists. Skipping duplicate." continue fi case "$TYPE" in casper) APPEND_LINE="APPEND initrd=$DIR/initrd boot=casper netboot=url url=http://${PI_IP}:8080/$ISO ip=dhcp" ;; arch) APPEND_LINE="APPEND initrd=$DIR/initrd archisobasedir=arch archiso_http_srv=http://${PI_IP}:8080/$DIR/ ip=dhcp" ;; debian-live) APPEND_LINE="APPEND initrd=$DIR/initrd boot=live fetch=http://${PI_IP}:8080/$ISO components ip=dhcp" ;; esac cat <> /srv/tftp/pxelinux.cfg/default LABEL $DIR MENU LABEL $NAME KERNEL $DIR/vmlinuz $APPEND_LINE $PASSWORD_LINE EOF success "Added $NAME to boot menu" done # ==================== DNSMASQ CONFIG ==================== if [ "$SETUP_STATE" == "NEW" ]; then DNSMASQ_CONF="# PXE Server Configuration\nport=0\nenable-tftp\ntftp-root=/srv/tftp\nlog-dhcp\n\ndhcp-range=${SUBNET},proxy\npxe-service=x86PC, \"Network Boot\", pxelinux\ndhcp-boot=pxelinux.0, ${HOSTNAME}, ${PI_IP}" if [[ "$ENABLE_MAC_FILTER" == "y" || "$ENABLE_MAC_FILTER" == "Y" ]]; then DNSMASQ_CONF+="\n\ndhcp-ignore=tag:!known" for mac in "${MAC_LIST[@]}"; do DNSMASQ_CONF+="\ndhcp-host=${mac},set:known"; done fi echo -e "$DNSMASQ_CONF" > /etc/dnsmasq.conf systemctl restart dnsmasq systemctl enable dnsmasq fi echo -e "\n${GREEN}======================================================${NC}" echo -e "${GREEN} ALL DONE! PXE SERVER UPDATED SUCCESSFULLY 🚀${NC}" echo -e "${GREEN}======================================================${NC}" echo -e "Served from: http://$PI_IP:8080"
