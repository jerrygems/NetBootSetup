🧹${NC}" echo -e "${GREEN} Your server is clean.${NC}" echo -e "${GREEN}======================================================${NC}" exit 0 fi # ==================== SECURITY OPTIONS ==================== echo "" echo -e "${BOLD}${YELLOW}Security Configuration:${NC}" echo -e "${CYAN}[i] These features prevent unauthorized machines from booting off your server.${NC}" echo "" # --- MAC Address Whitelisting --- ENABLE_MAC_FILTER="n" MAC_LIST=() echo -e "${BOLD}MAC Address Whitelisting${NC}" echo " Only laptops whose MAC addresses you whitelist will be allowed to PXE boot." echo " All other machines on the network will be completely ignored." read -p "Enable MAC whitelisting? Check internet or mirror at: $ISO_URL" verify_file "/var/www/html/$ISO_FILE" "$ISO_FILE" || die "ISO file missing after download" # Check ISO is not suspiciously tiny (sign of a 404 HTML page saved as .iso) ISO_SIZE=$(stat -c%s "/var/www/html/$ISO_FILE" 2>/dev/null || echo "0") if [ "$ISO_SIZE" -lt 100000000 ]; then die "Downloaded ISO is only $(du -h /var/www/html/$ISO_FILE | awk '{print $1}'). ISO might be malformed." fi verify_file /srv/tftp/$DIR/$INITRD_NAME "initramfs ($INITRD_NAME)" fi umount /mnt/iso 2>/dev/null || true # ==================== STEP 6: CONFIGURE SERVICES ==================== echo "" echo -e "${BLUE}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}" echo -e "${BLUE} STEP 6/6: Configuring dnsmasq & Boot Menu${NC}" echo -e "${BLUE}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}" # --- Build dnsmasq.conf --- if [ !
